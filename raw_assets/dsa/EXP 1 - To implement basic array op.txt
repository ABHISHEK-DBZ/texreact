(4) EXP 6 :- To convert an infix expression into its prefix (Polish) form using a stack.

CODE :-

def precedence(op):
    if op == '^':
        return 3
    elif op in '*/%':
        return 2
    elif op in '+-':
        return 1
    return 0


def infix_to_prefix(expression):
    expression = expression[::-1]

    expression = ''.join(
        ')' if c == '(' else '(' if c == ')' else c
        for c in expression
    )

    stack = []
    result = ""

    for char in expression:
        if char.isalnum():
            result += char

        elif char == '(':
            stack.append(char)

        elif char == ')':
            while stack and stack[-1] != '(':
                result += stack.pop()

            if stack:
                stack.pop()

        else:
            while (
                stack
                and stack[-1] != '('
                and precedence(stack[-1]) > precedence(char)
            ):
                result += stack.pop()

            stack.append(char)

    while stack:
        result += stack.pop()

    return result[::-1]


expression = input("Enter infix expression: ")

print("Prefix expression:", infix_to_prefix(expression))


Algorithm :-

1. Reverse the infix expression and interchange '(' and ')'.
2. Scan the characters left to right. If it is an operand (letter/digit), append it to the result. 
3. If it is '(' push it on the stack.
4. If it is ')' pop operators to the result until '(' is found, then discard '('. 
5. If it is an operator, pop operators of higher precedence from the stack to the result, then push the current operator. 
6. After the scan, pop all remaining operators to the result. 
7. Reverse the result; this is the prefix expression.

def insertion_sort(arr):
    comparisons = 0
    shifts = 0
    for i in range(1, len(arr)):
        key = arr[i]
        j= i -1
        print(f"\nComparing key = {key}")
        while j >= 0:
            comparisons += 1
            print(f"Compare {arr[j]} > {key}")
            if arr[j] > key:
                shifts += 1
                print(f"Shift {arr[j]} from index {j} to index {j+1}")
                arr[j + 1] = arr[j]
                j -= 1
            else:
                print(f"No shift: {arr[j]} <= {key}")
                break
        arr[j + 1] = key
        print(f"Insert {key} at index {j +1}")
        print("Array:", arr)
    return arr, comparisons, shifts
arr = list(map(int, input("Enter elements separated by spaces: ").split()))
arr, comparisons, shifts = insertion_sort(arr)

print("\nSorted array in ascending order:", arr)
print("Number of comparisons:", comparisons)
print("Number of shifts:", shifts)


Algorithm :- 

1) Start.

2) Read the number of elements \(n\) and the array.

3) Start from the second element, setting \(i = 1\).

4) Store the current element in key and set \(j = i - 1\).

5) Compare key with the preceding elements.

6) Shift elements greater than key one position to the right.

7) Insert key at its correct position.

8) Repeat steps 4–7 until all elements are processed.

9) Stop.

class Node:
    def __init__(self, key):
        self.val = key
        self.left = None
        self.right = None


class BinarySearchTree:
    def __init__(self):
        self.root = None

    # Insertion Operation
    def insert(self, key):
        self.root = self._insert_recursive(self.root, key)

    def _insert_recursive(self, node, key):
        if node is None:
            return Node(key)

        if key < node.val:
            node.left = self._insert_recursive(node.left, key)
        elif key > node.val:
            node.right = self._insert_recursive(node.right, key)

        return node

    # Deletion Operation
    def delete(self, key):
        self.root = self._delete_recursive(self.root, key)

    def _delete_recursive(self, node, key):
        if node is None:
            return node

        if key < node.val:
            node.left = self._delete_recursive(node.left, key)
        elif key > node.val:
            node.right = self._delete_recursive(node.right, key)
        else:
            if node.left is None:
                return node.right
            elif node.right is None:
                return node.left

            # Node with two children: Get the inorder successor
            temp = self._min_value_node(node.right)
            node.val = temp.val
            node.right = self._delete_recursive(node.right, temp.val)

        return node

    def _min_value_node(self, node):
        current = node

        while current.left is not None:
            current = current.left

        return current

    # Traversal Operations
    def inorder(self):
        res = []
        self._inorder_rec(self.root, res)
        return res

    def _inorder_rec(self, node, res):
        if node:
            self._inorder_rec(node.left, res)
            res.append(node.val)
            self._inorder_rec(node.right, res)

    def preorder(self):
        res = []
        self._preorder_rec(self.root, res)
        return res

    def _preorder_rec(self, node, res):
        if node:
            res.append(node.val)
            self._preorder_rec(node.left, res)
            self._preorder_rec(node.right, res)

    def postorder(self):
        res = []
        self._postorder_rec(self.root, res)
        return res

    def _postorder_rec(self, node, res):
        if node:
            self._postorder_rec(node.left, res)
            self._postorder_rec(node.right, res)
            res.append(node.val)

Algorithm :- 

A) Algorithm for Insertion

1) Start.

2) Create a new node and store the required data.

3) If the tree is empty, make the new node the root.

4) Otherwise, compare the new value with the current node.

5) If the new value is smaller, move to the left subtree.

6) If the new value is greater, move to the right subtree.

7) Repeat steps 5–6 until an empty position is found.

8) Insert the new node at that position.

9) Stop.


B. Algorithm for Deletion

1) Start.

2) Search for the node to be deleted.

3) If the value is smaller than the current node, search in the left subtree.

4) If the value is greater, search in the right subtree.

5) If the node is found, check its children.

6) Case 1: If the node has no child, delete the node.

7) Case 2: If the node has one child, replace the node with its child.

8) Case 3: If the node has two children, find the inorder successor and replace the node with it.

9) Stop.


C. Algorithm for Inorder Traversal
1) Start.

2) If the root is NULL, return.

3) Traverse the left subtree.

4) Visit (print) the root node.

5) Traverse the right subtree.

6) Stop.


D. Algorithm for Preorder Traversal
1) Start.

2) If the root is NULL, return.

3) Visit (print) the root node.

4) Traverse the left subtree.

5) Traverse the right subtree.

6) Stop.


(8) EXP 9 :- To Implement a Breadth First Search (BFS) Traversal on a graph and display the vertices visited during the traversal.

from collections import deque


class Graph:
    def __init__(self):
        self.graph = {}

    def add_vertex(self, vertex):
        if vertex not in self.graph:
            self.graph[vertex] = []

    def add_edge(self, source, destination):
        self.add_vertex(source)
        self.add_vertex(destination)

        if destination not in self.graph[source]:
            self.graph[source].append(destination)

        if source not in self.graph[destination]:
            self.graph[destination].append(source)

    def bfs(self, start_vertex):
        if start_vertex not in self.graph:
            print(f"Error: Vertex '{start_vertex}' not found in the graph.")
            return

        visited = set()
        queue = deque()

        visited.add(start_vertex)
        queue.append(start_vertex)

        print("\n--- BFS Traversal Output ---")
        print("Visited Order:", end=" ")

        while queue:
            current_vertex = queue.popleft()
            print(current_vertex, end=" ")

            for neighbor in self.graph[current_vertex]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    queue.append(neighbor)

        print("\n----------------------------")

    def display_bfs_tree(self, start_vertex):
        """Displays the BFS tree structure level-by-level."""

        if start_vertex not in self.graph:
            print(f"Error: Vertex '{start_vertex}' not found in the graph.")
            return

        visited = {start_vertex}
        queue = deque([start_vertex])

        levels = {start_vertex: 0}
        tree_structure = {}

        while queue:
            current_vertex = queue.popleft()
            current_level = levels[current_vertex]

            if current_level not in tree_structure:
                tree_structure[current_level] = []

            tree_structure[current_level].append(current_vertex)

            for neighbor in self.graph[current_vertex]:
                if neighbor not in visited:
                    visited.add(neighbor)
                    levels[neighbor] = current_level + 1
                    queue.append(neighbor)

        print("\n--- BFS Tree Structure (Level-by-Level) ---")
        print(f"Root Vertex: {start_vertex}")

        for level, nodes in tree_structure.items():
            print(f"Level {level}: {nodes}")

        print("------------------------------------------")

    def display_graph(self):
        print("\n--- Graph Adjacency List ---")

        if not self.graph:
            print("The graph is currently empty.")

        for vertex, neighbors in self.graph.items():
            print(f"Vertex {vertex} -> Connected to: {neighbors}")

        print("----------------------------")

Algorithm :- 

A. Algorithm for Adding an Edge

1) Start with the adjacency list adj and take the source u and destination v.

2) Append v to adj[u].

3) Append u to adj[v] (skip this step for a directed graph).

4) Stop.

B. Algorithm for BFS Traversal

1) Input the graph and the source vertex s.

2) Create an empty queue and a visited array. Set all vertices to false.

3) Mark visited[s] = true and enqueue s.

4) While the queue is not empty:
- Dequeue the front vertex and display it.
- For each neighbour in adj[s], if visited[j] is false, mark it true and enqueue it.
5) Stop.
